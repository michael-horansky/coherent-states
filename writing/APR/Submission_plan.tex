	\section{Research to be completed}
	
	\begin{enumerate}
	
	\item Monte Carlo method for electronic ground state estimation
	
	\begin{enumerate}
	
		\item Validity of and extensions for LEGS for larger molecules
		
		\item Treatment of multireferentiality
	
	\end{enumerate}
	
	\item Potential energy extrapolation via coherent state transport
	
	\begin{enumerate}
	
		\item How to best adiabatically deform coherent states?
		
		\item Can we retrieve PESs with good precision by transferring a sample of states into which an FCI solution was decomposed at one point in the parameter space?
		
		\item How does the method fare when encountering surface crossings (degeneracy)?
		
		\item Multireferentiality, cloning... and other potentially improving ideas
	
	\end{enumerate}
	
	\item Theoretical analysis of Qubit states (a.k.a. ACPNCCS)
	
	\begin{enumerate}
	
		\item Topology of the Qubit state manifold
		
		\item Coherence of Qubit states under static Hamiltonian evolution
	
	\end{enumerate}
	
	\end{enumerate}
